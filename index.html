const SHEETS = { deposito: "Deposito", retiros: "Retiros" }; const HEADERS = { deposito: ["id","cliente","negocio","subtipo","transporte","botellas","pago","fechaIngreso","destino","notas","retirado","fechaSalida","pedidos","ubicacion","fechaPago","shipping"], retiros: ["id","cliente","zona","negocio","transportistaRetiro","fechaRetiro","retirado","ingresado","transportistaSalida","notas","esExcepcion","fechaCreacion","subpedidos"] }; function getOrCreateSheet(name, headers) { const ss = SpreadsheetApp.getActiveSpreadsheet(); let sh = ss.getSheetByName(name); if (!sh) { sh = ss.insertSheet(name); sh.appendRow(headers); } return sh; } function sheetToObjects(sh) { const vals = sh.getDataRange().getValues(); if (vals.length < 2) return []; const hdrs = vals[0]; return vals.slice(1).map(r => { const o = {}; hdrs.forEach((h,i) => o[h] = r[i]); return o; }); } function doGet(e) { const action = e.parameter.action || "get"; const target = e.parameter.target || "deposito"; const sh = getOrCreateSheet(SHEETS[target], HEADERS[target]); if (action === "get") { return json({ ok: true, data: sheetToObjects(sh) }); } if (action === "add") { const row = JSON.parse(e.parameter.row); sh.appendRow(HEADERS[target].map(h => row[h] !== undefined ? row[h] : "")); return json({ ok: true }); } if (action === "update") { const id = String(e.parameter.id); const field = e.parameter.field; const value = e.parameter.value; const vals = sh.getDataRange().getValues(); const idCol = vals[0].indexOf("id"); const fCol = vals[0].indexOf(field); for (let i = 1; i < vals.length; i++) { if (String(vals[i][idCol]) === id) { sh.getRange(i+1, fCol+1).setValue(value); break; } } return json({ ok: true }); } if (action === "delete") { const id = String(e.parameter.id); const vals = sh.getDataRange().getValues(); const idCol = vals[0].indexOf("id"); for (let i = 1; i < vals.length; i++) { if (String(vals[i][idCol]) === id) { sh.deleteRow(i+1); break; } } return json({ ok: true }); } return json({ ok: false, error: "unknown" }); } function doPost(e) { return doGet(e); } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)).setMimeType(ContentService.MimeType.JSON); }
